// Copy to apps/erp/test/<name>.test.ts, source the worktree .env files, then:
//   mise exec node@24 -- pnpm exec vitest run test/<name>.test.ts --silent=false
// It runs importCsvTable inside one transaction (savepoints) and rolls back.
// Times importCsvTable on 3,000 customers inside one rolled-back transaction.
import {
  type DatabaseConnection,
  type Driver,
  Kysely,
  PostgresAdapter,
  PostgresIntrospector,
  PostgresQueryCompiler
} from "kysely";
import pg from "pg";
import { createHash } from "node:crypto";
import { importCsvTable } from "~/modules/shared/shared.service";

const COMPANY = "dbkl37nt40gg2cavfk6g";
const USER = "f2f45487-a4f5-4681-9c50-58959bfd9fbc";
const TERM = "pterm_JEB2x46trZzbQ5FsJ18Gzf";
const N = 3000;

import { it } from "vitest";
it("csv import bench", async () => {
const client = new pg.Client({ connectionString: "postgresql://postgres:postgres@localhost:55478/postgres" });
await client.connect();
await client.query("BEGIN");
let statements = 0;
let depth = 0;
const connection: DatabaseConnection = {
  async executeQuery(cq) {
    statements++;
    const r = await client.query(cq.sql, [...cq.parameters]);
    return { rows: r.rows, numAffectedRows: r.rowCount == null ? undefined : BigInt(r.rowCount) };
  },
  // biome-ignore lint/correctness/useYield: not used
  async *streamQuery() { throw new Error("no streaming"); }
};
const driver: Driver = {
  async init() {},
  async acquireConnection() { return connection; },
  async beginTransaction() { depth++; await client.query(`SAVEPOINT sp${depth}`); },
  async commitTransaction() { await client.query(`RELEASE SAVEPOINT sp${depth}`); depth--; },
  async rollbackTransaction() { await client.query(`ROLLBACK TO SAVEPOINT sp${depth}`); depth--; },
  async releaseConnection() {},
  async destroy() {}
};
const db = new Kysely<any>({
  dialect: {
    createAdapter: () => new PostgresAdapter(),
    createDriver: () => driver,
    createIntrospector: (k) => new PostgresIntrospector(k),
    createQueryCompiler: () => new PostgresQueryCompiler()
  }
});

const csv = (city: string) =>
  ["id,name,locationName,addressLine1,city,postalCode,countryCode,paymentTermId",
   ...Array.from({ length: N }, (_, i) => `bench-${i},Bench Customer ${i},HQ ${i},${i} Main St,${city} ${i % 50},${10000 + i},US,${TERM}`)
  ].join("\n");

let file = "";
const supabase = {
  storage: {
    from: () => ({ download: async () => ({ data: new Blob([file]), error: null }) })
  }
};
const ctx = { asSystem: { pg: db, supabase }, company: { id: COMPANY }, auditUserId: USER } as never;
const columnMappings = { id: "id", name: "name", locationName: "locationName", addressLine1: "addressLine1", city: "city", postalCode: "postalCode", countryCode: "countryCode", paymentTermId: "paymentTermId" };
const run = async (label: string, city: string) => {
  file = csv(city);
  statements = 0;
  const t0 = performance.now();
  const result = await importCsvTable(ctx, { table: "customer", filePath: `${COMPANY}/imports/bench.csv`, columnMappings, enumMappings: { paymentTermId: { [TERM]: TERM } } } as never);
  console.log(label, `${(performance.now() - t0).toFixed(0)} ms`, `${statements} statements`, JSON.stringify(result).slice(0, 160));
};
try {
  await run("insert", "Springfield");
  await run("update", "Shelbyville");
  const { rows } = await client.query(`
    SELECT c.name, cl.name AS loc, a."addressLine1", a.city, a."postalCode", a."countryCode", cp."paymentTermId"
    FROM customer c
    LEFT JOIN "customerLocation" cl ON cl."customerId" = c.id
    LEFT JOIN address a ON a.id = cl."addressId"
    LEFT JOIN "customerPayment" cp ON cp."customerId" = c.id
    WHERE c."companyId" = $1 AND c.name LIKE 'Bench Customer %'
    ORDER BY c.name, cl.name`, [COMPANY]);
  console.log("rows", rows.length, "md5", createHash("md5").update(JSON.stringify(rows)).digest("hex"), JSON.stringify(rows[0]));
} finally {
  await client.query("ROLLBACK");
  await client.end();
}

}, 600000);
