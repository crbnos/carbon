// Prints the queries that repeat (Q lines), the statement count and row digests.
// Statement count and digests for completing an operation with 20 materials.
import { sql } from "kysely";
import { it } from "vitest";
import { issueMaterial as issue } from "../../app/modules/inventory/inventory.service";
import { invokePosting } from "../../app/modules/shared/shared.service";
import { alwaysPostFixture, jobFixture, stampCutover, stockPart, USER, unwrap } from "./fixtures/always-post";
import { handlesOf } from "./fixtures/handles";

it("issue bench", async () => {
  const f = await alwaysPostFixture();
  try {
    await stampCutover(f);
    await stockPart(f, 1000);
    const job = await jobFixture(f, "1");
    const material = await f.db.selectFrom("jobMaterial").selectAll().where("id", "=", job.materialId).executeTakeFirstOrThrow();
    await f.db.insertInto("jobMaterial").values(Array.from({ length: 19 }, (_, i) => ({ jobId: material.jobId, jobMakeMethodId: material.jobMakeMethodId, jobOperationId: material.jobOperationId, itemId: material.itemId, itemType: "Part" as const, methodType: "Pull from Inventory" as const, description: `M${i}`, quantity: 1 + (i % 3), estimatedQuantity: 1 + (i % 3), unitOfMeasureCode: "EA", companyId: f.companyId, createdBy: USER }))).execute();
    const calls = async () => Number((await sql<{ n: string }>`SELECT sum(calls) AS n FROM pg_stat_statements`.execute(f.db)).rows[0]!.n);
    const snap = async () => new Map((await sql<{ q: string; c: string }>`SELECT queryid::text || ' ' || left(regexp_replace(query, '\s+', ' ', 'g'), 150) AS q, calls::text AS c FROM pg_stat_statements`.execute(f.db)).rows.map((r) => [r.q, Number(r.c)]));
    const s0 = await snap();
    const before = await calls();
    const t0 = performance.now();
    unwrap(await invokePosting(handlesOf(f.ctx), "issue", { type: "jobOperation", id: job.operationId, quantity: 1 }, issue));
    const s1 = await snap();
    const deltas = [...s1].map(([q, c]) => [q, c - (s0.get(q) ?? 0)] as const).filter(([, d]) => d > 0).sort((x, y) => y[1] - x[1]).slice(0, 18);
    for (const [q, d] of deltas) console.log("Q", d, q.replace(/^\S+ /, ""));
    console.log("issue", `${(performance.now() - t0).toFixed(0)} ms`, `${(await calls()) - before - 1} statements`);
    for (const table of ["itemLedger", "costLedger", "journalLine", "jobMaterial", "pickMethod", "itemCost"]) {
      const { rows } = await sql<{ n: number; h: string }>`
        SELECT count(*)::int AS n, md5(coalesce(string_agg(r, '|' ORDER BY r), '')) AS h FROM (
          SELECT replace((SELECT jsonb_object_agg(k, v) FROM jsonb_each(to_jsonb(t)) e(k, v)
            WHERE k !~ '(^id$|Id$|At$|By$|^entryNumber$|Date$|^journalEntryId$|^readable|Reference$)' OR k IN ('itemId', 'locationId', 'accountId', 'defaultStorageUnitId'))::text, ${f.prefix}, 'P') AS r
          FROM ${sql.table(table)} t WHERE t."companyId" = ${f.companyId}) x`.execute(f.db);
      console.log(table, rows[0]!.n, rows[0]!.h);
    }
  } finally {
    await f.cleanup();
  }
}, 600000);
