// It runs importCsvTable inside one transaction (savepoints) and rolls back.
// Times importCsvTable on 3,000 customers inside one rolled-back transaction.
import {
  type DatabaseConnection,
  type Driver,
  Kysely,
  PostgresAdapter,
  PostgresIntrospector,
  PostgresQueryCompiler
} from "kysely";
import pg from "pg";
import { createHash } from "node:crypto";
import { importCsvTable } from "~/modules/shared/shared.service";

const COMPANY = "dbkl37nt40gg2cavfk6g";
const USER = "f2f45487-a4f5-4681-9c50-58959bfd9fbc";
const TERM = "pterm_JEB2x46trZzbQ5FsJ18Gzf";
const N = 3000;

import { it } from "vitest";
it("csv import bench", async () => {
const client = new pg.Client({ connectionString: "postgresql://postgres:postgres@localhost:55478/postgres" });
await client.connect();
await client.query("BEGIN");
let statements = 0;
let depth = 0;
const connection: DatabaseConnection = {
  async executeQuery(cq) {
    statements++;
    const r = await client.query(cq.sql, [...cq.parameters]);
    return { rows: r.rows, numAffectedRows: r.rowCount == null ? undefined : BigInt(r.rowCount) };
  },
  // biome-ignore lint/correctness/useYield: not used
  async *streamQuery() { throw new Error("no streaming"); }
};
const driver: Driver = {
  async init() {},
  async acquireConnection() { return connection; },
  async beginTransaction() { depth++; await client.query(`SAVEPOINT sp${depth}`); },
  async commitTransaction() { await client.query(`RELEASE SAVEPOINT sp${depth}`); depth--; },
  async rollbackTransaction() { await client.query(`ROLLBACK TO SAVEPOINT sp${depth}`); depth--; },
  async releaseConnection() {},
  async destroy() {}
};
const db = new Kysely<any>({
  dialect: {
    createAdapter: () => new PostgresAdapter(),
    createDriver: () => driver,
    createIntrospector: (k) => new PostgresIntrospector(k),
    createQueryCompiler: () => new PostgresQueryCompiler()
  }
});

const PARTS = "ADCS-001,ANT-PATCH-01,BAT-LIION-48V,BRG-6201,BUS-STR-001,COMMS-001,EPS-001,FST-M4-TI,FST-M6-A286,HARNESS-001,PCB-ADCS-R1,PCB-BARE-REV3,PCB-EPS-R1,PROP-001,RW-010,SAT-1000,SAW-001,ST-050,TANK-TI-4L,THR-HYDRA-1N,TXRX-SBAND,VLV-SOLENOID-LP,VLV-SOLENOID-LP2".split(",");
const MATS = "MAT-AL7075-PLT,MAT-CF-LAM,MAT-GAAS-CELL,MAT-KAPTON,MAT-SYLGARD".split(",");
const PROCS = ["Machining", "Welding", "Clean Room Assembly"];
const bomCsv = ["parentId,componentId,quantity", ...PARTS.flatMap((p) => MATS.map((m, i) => `${p},${m},${i + 1}`))].join("\n");
const opsCsv = ["rowType,parentId,opNo,process,operationDescription,setupTime,setupUnit,laborUnit,machineUnit,stepName,stepType,toolId,toolQuantity,paramKey,paramValue",
  ...PARTS.flatMap((p) => Array.from({ length: 10 }, (_, k) => {
    const op = 900 + k;
    return [
      `BOP,${p},${op},${PROCS[k % 3]},Bench op ${k},${k},Total Minutes,Minutes/Piece,Minutes/Piece,,,,,,`,
      ...[1, 2, 3].map((s) => `STEP,${p},${op},,,,,,,Step ${s},Task,,,,`),
      `TOOL,${p},${op},,,,,,,,,${k % 2 ? "TL-TORQUE-J1" : "TL-PROBE-VNA"},${k + 1},,`,
      ...["torque", "speed"].map((key) => `PARAM,${p},${op},,,,,,,,,,,${key},${k}`)
    ];
  }).flat())].join("\n");

let file = "";
const supabase = {
  storage: {
    from: () => ({ download: async () => ({ data: new Blob([file]), error: null }) })
  }
};
const ctx = { asSystem: { pg: db, supabase }, company: { id: COMPANY }, auditUserId: USER } as never;
const ident = (xs: string[]) => Object.fromEntries(xs.map((x) => [x, x]));
const run = async (label: string, table: string, csvText: string, columns: string[], enumMappings: Record<string, Record<string, string>>) => {
  file = csvText;
  statements = 0;
  const t0 = performance.now();
  const result = await importCsvTable(ctx, { table, filePath: `${COMPANY}/imports/bench.csv`, columnMappings: ident(columns), enumMappings } as never);
  console.log(label, `${(performance.now() - t0).toFixed(0)} ms`, `${statements} statements`, JSON.stringify(result).slice(0, 300), label === "operations" ? JSON.stringify({ added: (result as any).added, nErr: (result as any).errors.length, nSkip: (result as any).skipped.length, sample: (result as any).skipped.slice(0, 2), warn: (result as any).warnings.slice(0, 2) }) : "");
};
try {
  for (const t of ["methodMaterial", "methodOperation", "methodOperationStep", "methodOperationTool", "methodOperationParameter"]) {
    await client.query(`CREATE TEMP TABLE "before_${t}" AS SELECT id FROM "${t}" WHERE "companyId" = $1`, [COMPANY]);
  }
  await run("bom", "bom", bomCsv, ["parentId", "componentId", "quantity"], {});
  await run("operations", "operations", opsCsv, ["rowType", "parentId", "opNo", "process", "operationDescription", "setupTime", "setupUnit", "laborUnit", "machineUnit", "stepName", "stepType", "toolId", "toolQuantity", "paramKey", "paramValue"], { rowType: ident(["BOP", "STEP", "TOOL", "PARAM"]), stepType: ident(["Task"]), setupUnit: ident(["Total Minutes"]), laborUnit: ident(["Minutes/Piece"]), machineUnit: ident(["Minutes/Piece"]) });
  const digest = async (q: string) => {
    const { rows } = await client.query(q, [COMPANY]);
    return `${rows.length} ${createHash("md5").update(JSON.stringify(rows)).digest("hex")}`;
  };
  console.log("materials", await digest(`SELECT "makeMethodId","itemId",quantity,"order","methodType","materialMakeMethodId","storageUnitIds","unitOfMeasureCode" FROM "methodMaterial" WHERE "companyId"=$1 AND id NOT IN (SELECT id FROM "before_methodMaterial") ORDER BY 1,2,4`));
  console.log("operations", await digest(`SELECT "makeMethodId","order",description,"processId","setupTime","operationType" FROM "methodOperation" WHERE "companyId"=$1 AND id NOT IN (SELECT id FROM "before_methodOperation") ORDER BY 1,2`));
  for (const [t, cols] of [["methodOperationStep", `s.name, s."sortOrder", s.type`], ["methodOperationTool", `s."toolId", s.quantity`], ["methodOperationParameter", `s.key, s.value`]] as const) {
    console.log(t, await digest(`SELECT o."makeMethodId", o."order", ${cols} FROM "${t}" s JOIN "methodOperation" o ON o.id = s."operationId" WHERE s."companyId"=$1 AND s.id NOT IN (SELECT id FROM "before_${t}") ORDER BY 1,2,3`));
  }
} finally {
  await client.query("ROLLBACK");
  await client.end();
}
}, 600000);
