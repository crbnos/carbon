// Statement counts and result digests for posting a 50-line shipment and receipt.
import { sql } from "kysely";
import { it } from "vitest";
import { postReceipt, postShipment } from "../../app/modules/inventory/inventory.service";
import { invokeCreate, invokePosting } from "../../app/modules/shared/shared.service";
import { alwaysPostFixture, seed, stampCutover, stockPart, USER, unwrap } from "./fixtures/always-post";
import { handlesOf } from "./fixtures/handles";

const N = 50;

it("posting bench", async () => {
  const f = await alwaysPostFixture();
  try {
    await stampCutover(f);
    await stockPart(f, 1000);
    const calls = async () =>
      Number((await sql<{ n: string }>`SELECT sum(calls) AS n FROM pg_stat_statements`.execute(f.db)).rows[0]!.n);
    const measure = async (label: string, run: () => Promise<unknown>) => {
      const before = await calls();
      const t0 = performance.now();
      await run();
      const ms = performance.now() - t0;
      console.log(label, `${ms.toFixed(0)} ms`, `${(await calls()) - before - 1} statements`);
    };

    const salesOrderId = `${f.prefix}-so-bench`;
    await seed(f, async (trx) => {
      await trx.insertInto("salesOrder").values({ id: salesOrderId, salesOrderId: "SO-BENCH", customerId: f.customerId, currencyCode: "USD", exchangeRate: 1, locationId: f.locationId, companyId: f.companyId, createdBy: USER }).execute();
      await trx.insertInto("salesOrderShipment").values({ id: salesOrderId, companyId: f.companyId }).execute();
      await trx.insertInto("salesOrderLine").values(Array.from({ length: N }, (_, i) => ({ salesOrderId, salesOrderLineType: "Part" as const, itemId: f.partId, saleQuantity: 1 + (i % 4), unitPrice: 25 + i, methodType: "Pull from Inventory" as const, unitOfMeasureCode: "EA", locationId: f.locationId, companyId: f.companyId, createdBy: USER }))).execute();
    });
    const shipment = unwrap(await invokeCreate(handlesOf(f.ctx), { type: "shipmentFromSalesOrder", salesOrderId, locationId: f.locationId }));
    await measure("postShipment", async () => unwrap(await invokePosting(handlesOf(f.ctx), "post-shipment", { type: "post", shipmentId: shipment.id }, postShipment)));

    const purchaseOrderId = `${f.prefix}-po-bench`;
    await seed(f, async (trx) => {
      const interaction = await trx.insertInto("supplierInteraction").values({ supplierId: f.supplierId, companyId: f.companyId }).returning("id").executeTakeFirstOrThrow();
      await trx.insertInto("purchaseOrder").values({ id: purchaseOrderId, purchaseOrderId: "PO-BENCH", supplierId: f.supplierId, supplierInteractionId: interaction.id, currencyCode: "USD", exchangeRate: 1, companyId: f.companyId, createdBy: USER }).execute();
      await trx.insertInto("purchaseOrderDelivery").values({ id: purchaseOrderId, locationId: f.locationId, companyId: f.companyId }).execute();
      await trx.insertInto("purchaseOrderLine").values(Array.from({ length: N }, (_, i) => ({ purchaseOrderId, purchaseOrderLineType: "Part" as const, itemId: f.partId, purchaseQuantity: 1 + (i % 5), supplierUnitPrice: 10 + i, exchangeRate: 1, conversionFactor: 1, inventoryUnitOfMeasureCode: "EA", locationId: f.locationId, companyId: f.companyId, createdBy: USER }))).execute();
    });
    const receipt = unwrap(await invokeCreate(handlesOf(f.ctx), { type: "receiptFromPurchaseOrder", purchaseOrderId, locationId: f.locationId }));
    await measure("postReceipt", async () => unwrap(await invokePosting(handlesOf(f.ctx), "post-receipt", { type: "post", receiptId: receipt.id }, postReceipt)));

    for (const table of ["itemLedger", "costLedger", "journalLine", "journal", "salesOrderLine", "salesOrder", "purchaseOrderLine", "purchaseOrder", "shipmentLine", "receiptLine", "shipment", "receipt", "itemCost"]) {
      const { rows } = await sql<{ n: number; h: string }>`
        SELECT count(*)::int AS n, md5(coalesce(string_agg(r, '|' ORDER BY r), '')) AS h FROM (
          SELECT replace((SELECT jsonb_object_agg(k, v) FROM jsonb_each(to_jsonb(t)) e(k, v)
            WHERE k !~ '(^id$|Id$|At$|By$|^entryNumber$|Date$|^journalEntryId$|^readable|Reference$)' OR k IN ('itemId', 'locationId', 'customerId', 'supplierId', 'accountId'))::text, ${f.prefix}, 'P') AS r
          FROM ${sql.table(table)} t WHERE t."companyId" = ${f.companyId}) x`.execute(f.db);
      console.log(table, rows[0]!.n, rows[0]!.h);
    }
  } finally {
    await f.cleanup();
  }
}, 600000);
