// NOSTEPS=1 skips the seeded method steps. Runs getMethod in one rolled-back transaction.
import {
  type DatabaseConnection,
  type Driver,
  Kysely,
  PostgresAdapter,
  PostgresIntrospector,
  PostgresQueryCompiler
} from "kysely";
import pg from "pg";
import { createHash } from "node:crypto";
import { getMethod } from "~/modules/production/production.service";

const COMPANY = "dbkl37nt40gg2cavfk6g";
const USER = "f2f45487-a4f5-4681-9c50-58959bfd9fbc";
const TERM = "pterm_JEB2x46trZzbQ5FsJ18Gzf";
const N = 3000;

import { it } from "vitest";
it("get method bench", async () => {
const client = new pg.Client({ connectionString: "postgresql://postgres:postgres@localhost:55478/postgres" });
await client.connect();
await client.query("BEGIN");
let statements = 0;
let depth = 0;
const connection: DatabaseConnection = {
  async executeQuery(cq) {
    statements++;
    const r = await client.query(cq.sql, [...cq.parameters]);
    return { rows: r.rows, numAffectedRows: r.rowCount == null ? undefined : BigInt(r.rowCount) };
  },
  // biome-ignore lint/correctness/useYield: not used
  async *streamQuery() { throw new Error("no streaming"); }
};
const driver: Driver = {
  async init() {},
  async acquireConnection() { return connection; },
  async beginTransaction() { depth++; await client.query(`SAVEPOINT sp${depth}`); },
  async commitTransaction() { await client.query(`RELEASE SAVEPOINT sp${depth}`); depth--; },
  async rollbackTransaction() { await client.query(`ROLLBACK TO SAVEPOINT sp${depth}`); depth--; },
  async releaseConnection() {},
  async destroy() {}
};
const db = new Kysely<any>({
  dialect: {
    createAdapter: () => new PostgresAdapter(),
    createDriver: () => driver,
    createIntrospector: (k) => new PostgresIntrospector(k),
    createQueryCompiler: () => new PostgresQueryCompiler()
  }
});

const ITEM = "item_LbiRz5c9U6W5ccyHgygwbc";
const JOB = "job_F75ZLT9K5zQPoMaumhKSwG";
const QUOTE_LINE = "quote_XHwh9WVMFtPUcaEz98Veo1:Y1PQCdcuRW8JL7iyx6jBHB";
const ctx = { asSystem: { pg: db }, company: { id: COMPANY }, auditUserId: USER } as never;
const digest = async (label: string, q: string) => {
  const { rows } = await client.query(`SELECT count(*)::int n, md5(coalesce(string_agg(r::text, '|' ORDER BY r::text), '')) h FROM (${q}) r`);
  console.log(label, rows[0].n, rows[0].h);
};
const jop = (o: string) => `(SELECT m."itemId" FROM "jobMakeMethod" m WHERE m.id = ${o}."jobMakeMethodId") || '#' || ${o}."order"`;
const qop = (o: string) => `(SELECT m."itemId" FROM "quoteMakeMethod" m WHERE m.id = ${o}."quoteMakeMethodId") || '#' || ${o}."order"`;
try {
  await client.query(`
    CREATE TEMP TABLE tree_ops AS
    WITH RECURSIVE tree AS (
      SELECT $1::text AS item, 1 AS lvl
      UNION
      SELECT mm."itemId", t.lvl + 1 FROM tree t
      JOIN "makeMethod" m ON m."itemId" = t.item AND m.status <> 'Archived'
      JOIN "methodMaterial" mm ON mm."makeMethodId" = m.id AND mm."methodType" = 'Make to Order'
      WHERE t.lvl < 6)
    SELECT o.* FROM "methodOperation" o JOIN "makeMethod" m ON m.id = o."makeMethodId" WHERE m."itemId" IN (SELECT item FROM tree)`, [ITEM]);
  await client.query(`INSERT INTO "methodOperation" SELECT (jsonb_populate_record(NULL::"methodOperation", to_jsonb(t) || jsonb_build_object('id', 'xop' || t.id, 'order', t."order" + 100, 'description', t.description || ' extra'))).* FROM tree_ops t`);
  await client.query(`INSERT INTO tree_ops SELECT * FROM "methodOperation" WHERE id LIKE 'xop%'`);
  const tool = (await client.query(`SELECT id FROM item WHERE "readableId" = 'TL-TORQUE-J1' AND "companyId" = $1`, [COMPANY])).rows[0].id;
  if (!process.env.NOSTEPS) await client.query(`INSERT INTO "methodOperationStep" (id, name, "sortOrder", type, "operationId", "companyId", "createdBy") SELECT 'xst' || s || o.id, 'Step ' || s, s, 'Task', o.id, $1, $2 FROM tree_ops o, generate_series(1, 3) s`, [COMPANY, USER]);
  await client.query(`INSERT INTO "methodOperationTool" (id, "operationId", "toolId", quantity, "companyId", "createdBy") SELECT 'xtl' || o.id, o.id, $3, 2, $1, $2 FROM tree_ops o`, [COMPANY, USER, tool]);
  if (!process.env.NOSTEPS) await client.query(`INSERT INTO "methodOperationToolStep" ("methodOperationToolId", "methodOperationStepId") SELECT 'xtl' || o.id, 'xst2' || o.id FROM tree_ops o`);
  await client.query(`INSERT INTO "methodOperationParameter" (id, key, value, "operationId", "companyId", "createdBy") SELECT 'xpm' || k || o.id, k, '7', o.id, $1, $2 FROM tree_ops o, unnest(ARRAY['torque','speed']) k`, [COMPANY, USER]);
  console.log("method operations in tree", (await client.query("SELECT count(*) FROM tree_ops")).rows[0].count);

  for (const [label, input] of [["itemToJob", { type: "itemToJob", sourceId: ITEM, targetId: JOB }], ["itemToQuoteLine", { type: "itemToQuoteLine", sourceId: ITEM, targetId: QUOTE_LINE }]] as const) {
    statements = 0;
    const t0 = performance.now();
    await getMethod(ctx, input as never);
    console.log(label, `${(performance.now() - t0).toFixed(0)} ms`, `${statements} statements`);
  }
  await digest("jobMakeMethod", `SELECT m."itemId", (SELECT "itemId" FROM "jobMaterial" WHERE id = m."parentMaterialId") p, m."quantityPerParent", m.version, m."itemScrapPercentage" FROM "jobMakeMethod" m WHERE m."jobId" = '${JOB}'`);
  await digest("jobOperation", `SELECT ${jop("o")} k, o.description, o."processId", o."workCenterId", o."setupTime", o."laborTime", o."machineTime", o."operationType", o."targetQuantity", o."operationQuantity", o."procedureId", o."assemblyInstructionId", o."laborRate", o."operationOrder", o."workInstruction" FROM "jobOperation" o WHERE o."jobId" = '${JOB}'`);
  for (const [t, cols] of [["jobOperationStep", `c.name, c."sortOrder", c.type, c."minValue", c."maxValue", c.description, c."assemblyInstructionStepId"`], ["jobOperationTool", `c."toolId", c.quantity`], ["jobOperationParameter", `c.key, c.value`]]) {
    await digest(t, `SELECT ${jop("o")} k, ${cols} FROM "${t}" c JOIN "jobOperation" o ON o.id = c."operationId" WHERE o."jobId" = '${JOB}'`);
  }
  await digest("jobOperationToolStep", `SELECT ${jop("o")} k, t."toolId", s.name FROM "jobOperationToolStep" ts JOIN "jobOperationTool" t ON t.id = ts."jobOperationToolId" JOIN "jobOperationStep" s ON s.id = ts."jobOperationStepId" JOIN "jobOperation" o ON o.id = t."operationId" WHERE o."jobId" = '${JOB}'`);
  await digest("jobMaterial", `SELECT (SELECT "itemId" FROM "jobMakeMethod" WHERE id = jm."jobMakeMethodId") mm, jm."itemId", jm."order", jm.quantity, jm."estimatedQuantity", jm."scrapQuantity", jm."methodType", jm."substitutedFromItemId", (SELECT "order" FROM "jobOperation" WHERE id = jm."jobOperationId") op FROM "jobMaterial" jm WHERE jm."jobId" = '${JOB}'`);
  for (const t of ["jobMaterialStep", "jobOperationStepSlide", "jobOperationDependency"]) {
    await digest(`${t} count`, `SELECT 1 FROM "${t}"`);
  }
  const [quoteId, quoteLineId] = QUOTE_LINE.split(":");
  await digest("quoteMakeMethod", `SELECT m."itemId", (SELECT "itemId" FROM "quoteMaterial" WHERE id = m."parentMaterialId") p, m."quantityPerParent", m.version FROM "quoteMakeMethod" m WHERE m."quoteLineId" = '${quoteLineId}'`);
  await digest("quoteOperation", `SELECT ${qop("o")} k, o.description, o."processId", o."workCenterId", o."setupTime", o."laborTime", o."machineTime", o."operationType", o."procedureId", o."laborRate", o."operationOrder", o."workInstruction" FROM "quoteOperation" o WHERE o."quoteLineId" = '${quoteLineId}'`);
  for (const [t, cols] of [["quoteOperationStep", `c.name, c."sortOrder", c.type, c."minValue", c."maxValue", c.description`], ["quoteOperationTool", `c."toolId", c.quantity`], ["quoteOperationParameter", `c.key, c.value`]]) {
    await digest(t, `SELECT ${qop("o")} k, ${cols} FROM "${t}" c JOIN "quoteOperation" o ON o.id = c."operationId" WHERE o."quoteLineId" = '${quoteLineId}'`);
  }
  await digest("quoteOperationToolStep", `SELECT ${qop("o")} k, t."toolId", s.name FROM "quoteOperationToolStep" ts JOIN "quoteOperationTool" t ON t.id = ts."quoteOperationToolId" JOIN "quoteOperationStep" s ON s.id = ts."quoteOperationStepId" JOIN "quoteOperation" o ON o.id = t."operationId" WHERE o."quoteLineId" = '${quoteLineId}'`);
  await digest("quoteMaterial", `SELECT (SELECT "itemId" FROM "quoteMakeMethod" WHERE id = qm."quoteMakeMethodId") mm, qm."itemId", qm."order", qm.quantity, qm."scrapQuantity", qm."methodType", (SELECT "order" FROM "quoteOperation" WHERE id = qm."quoteOperationId") op FROM "quoteMaterial" qm WHERE qm."quoteLineId" = '${quoteLineId}'`);
  for (const t of ["quoteMaterialStep", "quoteOperationStepSlide"]) {
    await digest(`${t} count`, `SELECT 1 FROM "${t}"`);
  }

  // The other flows. Each target is digested right after its flow, so targets can be reused.
  const BH = ["item_BH6534bAen1JTmhwQsps2p", "make_5EqvG4nQtCEz1QuS66jnqY"];
  const JX = ["item_Jx8TCRpyNcU1oikLAdmGwu", "make_AEJZ2NBSjJCw4fawHDzZy3"];
  const UV = ["item_UvPTi8648GWLSeEea76Mf5", "make_CsPy2zE6RdWTuntocSzYVf"];
  const QL2 = "quote_MRsPQyNG96jifcW8xLgCkr:XT2FTzypX9eK38JyaxXTAf";
  const SUB = "item_JQk1v4HNZAspCyCT52A1hU";
  const JOB2 = "job_benchscratch";
  const jobCols = (await client.query(`SELECT string_agg(format('%I', column_name), ',') c FROM information_schema.columns WHERE table_schema = 'public' AND table_name = 'job' AND is_generated = 'NEVER' AND column_name NOT IN ('id', 'jobId', 'itemId', 'status')`)).rows[0].c;
  await client.query(`INSERT INTO job (id, "jobId", "itemId", status, ${jobCols}) SELECT $2, 'JBENCH-1', $3, 'Draft', ${jobCols} FROM job WHERE id = $1`, [JOB, JOB2, BH[0]]);
  const jmm = (await client.query(`SELECT id FROM "jobMakeMethod" WHERE "jobId" = $1 AND "parentMaterialId" IS NOT NULL ORDER BY "itemId", id LIMIT 1`, [JOB])).rows[0].id;
  const qmm = (await client.query(`SELECT id FROM "quoteMakeMethod" WHERE "quoteLineId" = $1 AND "parentMaterialId" IS NOT NULL ORDER BY "itemId", id LIMIT 1`, [QUOTE_LINE.split(":")[1]])).rows[0].id;
  const wop = (o: string) => `(SELECT m."itemId" FROM "makeMethod" m WHERE m.id = ${o}."makeMethodId") || '#' || ${o}."order"`;
  const digestMethod = async (label: string, makeMethodId: string) => {
    await digest(`${label}: methodOperation`, `SELECT ${wop("o")} k, o.description, o."processId", o."workCenterId", o."setupTime", o."laborTime", o."operationType", o."procedureId", o."workInstruction" FROM "methodOperation" o WHERE o."makeMethodId" = '${makeMethodId}'`);
    for (const [t, cols] of [["methodOperationStep", `c.name, c."sortOrder", c.type, c."minValue", c.description`], ["methodOperationTool", `c."toolId", c.quantity`], ["methodOperationParameter", `c.key, c.value`]]) {
      await digest(`${label}: ${t}`, `SELECT ${wop("o")} k, ${cols} FROM "${t}" c JOIN "methodOperation" o ON o.id = c."operationId" WHERE o."makeMethodId" = '${makeMethodId}'`);
    }
    await digest(`${label}: methodMaterial`, `SELECT mm."itemId", mm."order", mm.quantity, mm."methodType" FROM "methodMaterial" mm WHERE mm."makeMethodId" = '${makeMethodId}'`);
  };
  const digestJob = async (label: string, job: string) => {
    await digest(`${label}: jobOperation`, `SELECT ${jop("o")} k, o.description, o."processId", o."setupTime", o."operationType", o."targetQuantity" FROM "jobOperation" o WHERE o."jobId" = '${job}'`);
    for (const [t, cols] of [["jobOperationStep", `c.name, c."sortOrder", c.type`], ["jobOperationTool", `c."toolId", c.quantity`], ["jobOperationParameter", `c.key, c.value`]]) {
      await digest(`${label}: ${t}`, `SELECT ${jop("o")} k, ${cols} FROM "${t}" c JOIN "jobOperation" o ON o.id = c."operationId" WHERE o."jobId" = '${job}'`);
    }
    await digest(`${label}: jobOperationToolStep`, `SELECT ${jop("o")} k, t."toolId", s.name FROM "jobOperationToolStep" ts JOIN "jobOperationTool" t ON t.id = ts."jobOperationToolId" JOIN "jobOperationStep" s ON s.id = ts."jobOperationStepId" JOIN "jobOperation" o ON o.id = t."operationId" WHERE o."jobId" = '${job}'`);
    await digest(`${label}: jobMaterial`, `SELECT (SELECT "itemId" FROM "jobMakeMethod" WHERE id = jm."jobMakeMethodId") mm, jm."itemId", jm."order", jm.quantity, jm."methodType" FROM "jobMaterial" jm WHERE jm."jobId" = '${job}'`);
  };
  const digestQuoteLine = async (label: string, ql: string) => {
    await digest(`${label}: quoteOperation`, `SELECT ${qop("o")} k, o.description, o."processId", o."setupTime" FROM "quoteOperation" o WHERE o."quoteLineId" = '${ql}'`);
    for (const [t, cols] of [["quoteOperationStep", `c.name, c."sortOrder"`], ["quoteOperationTool", `c."toolId", c.quantity`], ["quoteOperationParameter", `c.key, c.value`]]) {
      await digest(`${label}: ${t}`, `SELECT ${qop("o")} k, ${cols} FROM "${t}" c JOIN "quoteOperation" o ON o.id = c."operationId" WHERE o."quoteLineId" = '${ql}'`);
    }
  };
  const flows: [string, Record<string, unknown>, () => Promise<void>][] = [
    ["itemToJobMakeMethod", { type: "itemToJobMakeMethod", sourceId: SUB, targetId: jmm }, () => digestJob("itemToJobMakeMethod", JOB)],
    ["itemToQuoteMakeMethod", { type: "itemToQuoteMakeMethod", sourceId: SUB, targetId: qmm }, () => digestQuoteLine("itemToQuoteMakeMethod", QUOTE_LINE.split(":")[1]!)],
    ["quoteLineToQuoteLine", { type: "quoteLineToQuoteLine", sourceId: QUOTE_LINE, targetId: QL2 }, () => digestQuoteLine("quoteLineToQuoteLine", QL2.split(":")[1]!)],
    ["jobToItem", { type: "jobToItem", sourceId: JOB, targetId: BH[1] }, () => digestMethod("jobToItem", BH[1]!)],
    ["jobMakeMethodToItem", { type: "jobMakeMethodToItem", sourceId: jmm, targetId: UV[1] }, () => digestMethod("jobMakeMethodToItem", UV[1]!)],
    ["quoteLineToItem", { type: "quoteLineToItem", sourceId: QUOTE_LINE, targetId: JX[1] }, () => digestMethod("quoteLineToItem", JX[1]!)],
    ["quoteMakeMethodToItem", { type: "quoteMakeMethodToItem", sourceId: qmm, targetId: UV[1] }, () => digestMethod("quoteMakeMethodToItem", UV[1]!)],
    ["makeMethodToMakeMethod", { type: "makeMethodToMakeMethod", sourceId: "make_2rWY1VmwUQZkbrWmhAR93u", targetId: BH[1] }, () => digestMethod("makeMethodToMakeMethod", BH[1]!)],
    ["itemToItem", { type: "itemToItem", sourceId: ITEM, targetId: JX[0] }, () => digestMethod("itemToItem", JX[1]!)],
    ["jobToJob", { type: "jobToJob", sourceId: JOB, targetId: JOB2 }, () => digestJob("jobToJob", JOB2)],
    ["quoteLineToJob", { type: "quoteLineToJob", sourceId: QUOTE_LINE, targetId: JOB2 }, () => digestJob("quoteLineToJob", JOB2)]
  ];
  for (const [label, input, check] of flows) {
    statements = 0;
    const t0 = performance.now();
    await client.query("SAVEPOINT flow");
    try {
      await getMethod(ctx, input as never);
      await client.query("RELEASE SAVEPOINT flow");
      console.log(label, `${(performance.now() - t0).toFixed(0)} ms`, `${statements} statements`);
      await check();
    } catch (e) {
      await client.query("ROLLBACK TO SAVEPOINT flow");
      console.log(label, "FAILED", String((e as Error).message).slice(0, 160));
    }
  }
} finally {
  await client.query("ROLLBACK");
  await client.end();
}
}, 600000);
